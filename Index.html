โค้ด Google Apps Script (รองรับข้อมูลวิธีสแกน QR & Face) 

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  sheet.appendRow([
    data.date,
    data.empId,
    data.empName,
    data.dept,
    data.clockIn,
    data.clockOut || '-',
    data.scanMethod, // 'QR Code' หรือ 'สแกนใบหน้า'
    data.status,
    data.location,
    data.note || '-',
    new Date()
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}
